ПОЛНОЕ ОБУЧЕНИЕ: JavaScript для блока отзывов с шахматным порядком

/// НАПИСАННО ПО СТАРОЙ ВЕРСИИ КОДА ///

Общая структура

Этот скрипт создаёт динамический блок отзывов с:

    Двумя горизонтальными рядами (верхний и нижний)

    Шахматным порядком (видимые отзывы + пустые заглушки)

    Синхронной прокруткой через ползунок

    Счётчиком текущего отзыва

Важно: весь код обёрнут в IIFE (Immediately Invoked Function Expression)

js
(function () {
    // весь код здесь
})();

Что это значит: Функция создаётся и сразу же вызывается. Все переменные внутри неё не попадают в глобальную область видимости (не создают конфликтов с другими скриптами на странице).



ЧАСТЬ 1: ДАННЫЕ ОТЗЫВОВ

js
const reviewsData = [
    { name: "Анна К.", text: "Потрясающий сервис!...", date: "12.05.2025" },
    { name: "Михаил П.", text: "Быстро, качественно...", date: "03.04.2025" },
    // ... и так далее
];

Что это:
Компонент	                                Объяснение
const	                                    Объявляет константу (значение нельзя переназначить)
reviewsData	                                Название переменной (массив)
[...]	                                    Массив (список) объектов
{ name: "...", text: "...", date: "..." }	Объект с тремя свойствами

Структура одного отзыва:

js
{
    name: "Анна К.",      // имя автора (строка)
    text: "Текст...",     // текст отзыва (строка)
    date: "12.05.2025"    // дата (строка)
}

Где хранятся данные:
Прямо в скрипте (статичные). Позже заменим на загрузку из базы данных.



ЧАСТЬ 2: ЗАЩИТА ОТ XSS-АТАК (безопасность)

js
function escapeHtml(str) {
    const div = document.createElement('div');
    div.textContent = str;
    return div.innerHTML;
}

Что такое XSS?
Cross-Site Scripting — когда злоумышленник вставляет вредоносный код в текст отзыва (например, <script>alert('взлом')</script>).

Как работает escapeHtml:
Шаг	Код	                                        Что происходит
1	const div = document.createElement('div')	Создаём пустой <div> в памяти
2	div.textContent = str	                    Вставляем текст как обычный текст (безопасно)
3	return div.innerHTML	                    Забираем HTML-представление, где спецсимволы заменены

Пример преобразования:
Вход:   "<script>alert('xss')</script>"
Выход:  "&lt;script&gt;alert('xss')&lt;/script&gt;"

Браузер прочитает это как обычный текст, а не как исполняемый код.

Почему это важно:
// ПЛОХО (опасно):
card.innerHTML = `<div>${userInput}</div>`;  // если userInput = "<script>..."

// ХОРОШО (безопасно):
card.innerHTML = `<div>${escapeHtml(userInput)}</div>`;



ЧАСТЬ 3: СОЗДАНИЕ КАРТОЧКИ ОТЗЫВА

js
function createCard(r) {
    const card = document.createElement('div');
    card.className = 'reviewsWidget__comment';
    card.innerHTML = `
        <div class="reviewsWidget__comment-content">
            <div class="reviewsWidget__commentName">${escapeHtml(r.name)}</div>
            <div class="reviewsWidget__commentText">${escapeHtml(r.text)}</div>
        </div>
        <div class="reviewsWidget__commentDate">${escapeHtml(r.date)}</div>
    `;
    return card;
}

Пошаговый разбор:
Строка	                                    Что делает
function createCard(r)	                    Объявляет функцию, которая принимает объект отзыва r
const card = document.createElement('div')	Создаёт HTML-элемент <div>
card.className = 'reviewsWidget__comment'	Добавляет CSS-класс (для стилизации)
card.innerHTML = ...``	                    Заполняет внутреннее содержимое карточки


Структура создаваемой карточки:

html
<div class="reviewsWidget__comment">
    <div class="reviewsWidget__comment-content">
        <div class="reviewsWidget__commentName">Анна К.</div>
        <div class="reviewsWidget__commentText">Текст отзыва...</div>
    </div>
    <div class="reviewsWidget__commentDate">12.05.2025</div>
</div>

Зачем нужен escapeHtml:

js
${escapeHtml(r.name)}   // имя
${escapeHtml(r.text)}   // текст
${escapeHtml(r.date)}   // дата

Каждое поле экранируется перед вставкой в HTML.

Почему используется обратный апостроф (`):

js
`строка ${переменная} строка`

Это шаблонные строки (template literals). Позволяют вставлять переменные прямо в строку через ${...}.



ЧАСТЬ 4: ПОИСК КОНТЕЙНЕРОВ НА СТРАНИЦЕ

js
const container = document.querySelector('.section_Three .reviewsWidget');
if (!container) return;

const topFeed = container.querySelector('.reviewsWidget__feed--top');
const bottomFeed = container.querySelector('.reviewsWidget__feed--bottom');
const range = container.querySelector('.reviewsWidget__range');
const label = container.querySelector('.reviewsWidget__thumbLabel');
const counter = container.querySelector('.reviewsWidget__counter');

Что делает querySelector:
Ищет первый элемент на странице, соответствующий CSS-селектору.
Селектор	                    Что ищет
.section_Three .reviewsWidget	Элемент с классом reviewsWidget ВНУТРИ элемента с классом section_Three
.reviewsWidget__feed--top	    Элемент с классом reviewsWidget__feed--top
.reviewsWidget__range	        Элемент с классом reviewsWidget__range

Проверка if (!container) return:
Если контейнер не найден — скрипт останавливается (ничего не делает). Это защита от ошибок, если блок отзывов отсутствует на странице.

Что такое const:
Переменная, которую нельзя переназначить (но содержимое объекта/массива менять можно).



ЧАСТЬ 5: ПЕРЕМЕННЫЕ ДЛЯ ПРОКРУТКИ

js
let maxTop = 0, maxBottom = 0;

Что это	        Объяснение
let	            Переменная, которую можно изменять
maxTop	        Максимальное значение прокрутки для верхнего блока (в пикселях)
maxBottom	    Максимальное значение прокрутки для нижнего блока

Как вычисляется максимальная прокрутка:

js
maxTop = scrollWidth - clientWidth

    scrollWidth — полная ширина контента (все карточки)

    clientWidth — видимая ширина контейнера

Пример:
Контейнер шириной 400px, карточки на 1200px.
maxTop = 1200 - 400 = 800px (максимум, на сколько можно прокрутить влево)



ЧАСТЬ 6: ОБНОВЛЕНИЕ МАКСИМАЛЬНОЙ ПРОКРУТКИ

js
function updateMaxScroll() {
    if (topFeed) maxTop = Math.max(0, topFeed.scrollWidth - topFeed.clientWidth);
    if (bottomFeed) maxBottom = Math.max(0, bottomFeed.scrollWidth - bottomFeed.clientWidth);
    if (counter) counter.innerText = `0 / ${reviewsData.length}`;
}

Пошагово:
Строка	                    Что делает
if (topFeed)	            Проверяет, существует ли верхний блок (защита от ошибок)
topFeed.scrollWidth	        Полная ширина контента внутри блока
topFeed.clientWidth	        Видимая ширина блока
Math.max(0, ...)	        Если разница отрицательная, берёт 0 (нельзя прокрутить)
counter.innerText = ...	    Устанавливает текст счётчика

Пример работы Math.max(0, value):
value = 800 → Math.max(0, 800) → 800
value = -50 → Math.max(0, -50) → 0



ЧАСТЬ 7: СИНХРОНИЗАЦИЯ ПРОКРУТКИ

js
function syncScroll(percent) {
    const p = Math.min(1, Math.max(0, percent / 100));
    if (topFeed && maxTop > 0) topFeed.scrollTo({ left: p * maxTop, behavior: 'smooth' });
    if (bottomFeed && maxBottom > 0) bottomFeed.scrollTo({ left: p * maxBottom, behavior: 'smooth' });
    if (counter) {
        let idx = Math.floor(p * reviewsData.length);
        idx = Math.min(reviewsData.length - 1, Math.max(0, idx));
        counter.innerText = `${idx + 1} / ${reviewsData.length}`;
    }
}

Разбор каждого шага:
Строка	                        Объяснение
percent	                        Значение ползунка от 0 до 100
percent / 100	                Превращает проценты в дробь (0.0 ... 1.0)
Math.min(1, Math.max(0, ...))	Ограничивает значение между 0 и 1
topFeed.scrollTo(...)	        Прокручивает блок до позиции p * maxTop
behavior: 'smooth'	            Плавная анимация прокрутки


Вычисление индекса текущего отзыва:

js
let idx = Math.floor(p * reviewsData.length);

    p = 0.25, reviewsData.length = 10 → 0.25 * 10 = 2.5 → Math.floor(2.5) = 2

    Отсчёт с 0 → показываем "3 / 10"


Защита границ:

js
idx = Math.min(reviewsData.length - 1, Math.max(0, idx));

    Math.max(0, idx) — не меньше 0

    Math.min(9, ...) — не больше 9 (при 10 элементах)



ЧАСТЬ 8: ПОЗИЦИОНИРОВАНИЕ НАДПИСИ НА ПОЛЗУНКЕ

js
function updateLabelPos() {
    const percent = range.value / 100;
    const rect = range.getBoundingClientRect();
    const left = percent * rect.width;
    label.style.left = left + 'px';
}

Что делает:
Свойство	                    Объяснение
range.value	                    Текущее значение ползунка (0-100)
percent	                        Дробное значение (0.0-1.0)
range.getBoundingClientRect()	Получает размеры и позицию ползунка на странице
rect.width	                    Ширина ползунка в пикселях
percent * rect.width	        Позиция, где находится бегунок
label.style.left = left + 'px'	Сдвигает надпись на эту позицию

Зачем нужен getBoundingClientRect():
Возвращает точные координаты элемента на странице после всех CSS-преобразований.



ЧАСТЬ 9: ПОСТРОЕНИЕ БЛОКОВ (ШАХМАТНЫЙ ПОРЯДОК)

js
function build() {
    if (!topFeed || !bottomFeed) return;
    topFeed.innerHTML = '';
    bottomFeed.innerHTML = '';

    reviewsData.forEach((r, idx) => {
        // Верхний блок
        if (idx % 2 === 0) {
            topFeed.appendChild(createCard(r));
        } else {
            const empty = document.createElement('div');
            empty.className = 'reviewsWidget__comment reviewsWidget__comment--empty';
            empty.style.cssText = 'opacity: 0; pointer-events: none;';
            topFeed.appendChild(empty);
        }

        // Нижний блок (инверсия)
        if (idx % 2 === 1) {
            bottomFeed.appendChild(createCard(r));
        } else {
            const empty = document.createElement('div');
            empty.className = 'reviewsWidget__comment reviewsWidget__comment--empty';
            empty.style.cssText = 'opacity: 0; pointer-events: none;';
            bottomFeed.appendChild(empty);
        }
    });

    updateMaxScroll();
}

Разбор:
Строка	                                        Объяснение
topFeed.innerHTML = ''	                        Очищает блок (удаляет все старые карточки)
reviewsData.forEach((r, idx) => {...})	        Перебирает каждый отзыв. idx = индекс (0,1,2,3...)
idx % 2 === 0	                                Чётный индекс? (0,2,4,6...)
idx % 2 === 1	                                Нечётный индекс? (1,3,5,7...)


Шахматная логика:
Индекс (idx)	Верхний блок	    Нижний блок
0 (чётный)	    Отзыв	            Пустая заглушка
1 (нечётный)	Пустая заглушка	    Отзыв
2 (чётный)	    Отзыв	            Пустая заглушка
3 (нечётный)	Пустая заглушка	    Отзыв
...	и так далее	и так далее


Создание пустой заглушки:

js
const empty = document.createElement('div');
empty.className = 'reviewsWidget__comment reviewsWidget__comment--empty';
empty.style.cssText = 'opacity: 0; pointer-events: none;';
topFeed.appendChild(empty);

Свойство	                                        Зачем
className = '... reviewsWidget__comment--empty'	    Для CSS-стилизации
opacity: 0	                                        Делает блок невидимым
pointer-events: none	                            Мышь не реагирует на блок
appendChild(empty)	                                Добавляет блок в конец контейнера



ЧАСТЬ 10: ПОДПИСКА НА СОБЫТИЕ ПОЛЗУНКА

js
if (range) {
    range.addEventListener('input', (e) => {
        syncScroll(parseInt(e.target.value, 10));
        updateLabelPos();
    });
}

Что такое событие input:
Срабатывает каждый раз, когда пользователь двигает ползунок (даже на 1 пиксель).

Разбор:
Код	                                    Что делает
if (range)	                            Проверка, что ползунок существует
range.addEventListener('input', ...)	Назначает обработчик на событие
(e) => {...}	                        Стрелочная функция, получающая объект события
e.target.value	                        Текущее значение ползунка (строка)
parseInt(..., 10)	                    Превращает строку в целое число (основание 10)
syncScroll(...)	                        Прокручивает блоки
updateLabelPos()	                    Обновляет позицию надписи


Почему parseInt с основанием 10:

js
parseInt("05", 10) → 5  // правильно
parseInt("05") → 5      // тоже работает, но лучше указывать



ЧАСТЬ 11: СИНХРОНИЗАЦИЯ ПРИ РУЧНОЙ ПРОКРУТКЕ

js
function bindSync(a, b) {
    if (!a || !b) return;
    a.addEventListener('scroll', () => {
        const maxA = a.scrollWidth - a.clientWidth;
        if (maxA <= 0) return;
        const p = a.scrollLeft / maxA;
        if (range && document.activeElement !== range) {
            range.value = p * 100;
            updateLabelPos();
        }
        const maxB = b.scrollWidth - b.clientWidth;
        if (maxB > 0) b.scrollTo({ left: p * maxB, behavior: 'smooth' });
        if (counter) {
            let idx = Math.floor(p * reviewsData.length);
            idx = Math.min(reviewsData.length - 1, Math.max(0, idx));
            counter.innerText = `${idx + 1} / ${reviewsData.length}`;
        }
    });
}

Зачем это нужно:
Когда пользователь руками прокручивает один блок (мышкой или тачем), второй блок и ползунок должны синхронизироваться.

Разбор:
Код	                                                    Объяснение
function bindSync(a, b)	                                Принимает два блока: источник a и цель b
a.addEventListener('scroll', ...)	                    Слушает событие прокрутки блока a
const p = a.scrollLeft / maxA	                        Вычисляет процент прокрутки (0.0-1.0)
document.activeElement !== range	                    Проверка: пользователь НЕ трогает ползунок (чтобы избежать зацикливания)
range.value = p * 100	                                Синхронизирует значение ползунка
b.scrollTo({ left: p * maxB, behavior: 'smooth' })	    Прокручивает второй блок

Что такое document.activeElement:
Возвращает элемент, на котором сейчас фокус (например, ползунок, который таскают мышкой).



ЧАСТЬ 12: ОБРАБОТКА ИЗМЕНЕНИЯ РАЗМЕРА ОКНА
javascript

let resizeTimer;
window.addEventListener('resize', () => {
    clearTimeout(resizeTimer);
    resizeTimer = setTimeout(() => {
        updateMaxScroll();
        if (range) syncScroll(range.value);
        updateLabelPos();
    }, 100);
});

Зачем это нужно:

При изменении размера окна:

    Меняется clientWidth контейнеров

    Меняется максимальная прокрутка (maxTop, maxBottom)

    Нужно пересчитать всё заново

Техника "debounce" (задержка):
Строка	                    Объяснение
let resizeTimer	            Переменная для хранения таймера
clearTimeout(resizeTimer)	Отменяет предыдущий таймер
setTimeout(..., 100)	    Запускает новый таймер на 100мс

Зачем задержка:
Событие resize срабатывает очень часто (сотни раз за секунду). Без задержки скрипт будет тормозить. Мы ждём 100мс после последнего изменения окна и только потом обновляем.



ЧАСТЬ 13: ЗАПУСК ВСЕГО СКРИПТА
javascript

build();
setTimeout(() => {
    updateMaxScroll();
    if (range) {
        syncScroll(0);
        updateLabelPos();
        bindSync(topFeed, bottomFeed);
        bindSync(bottomFeed, topFeed);
    }
}, 80);

Что происходит:
Строка	                Когда	                Что делает
build()	                Сразу	                Создаёт карточки и пустые заглушки
setTimeout(..., 80)	    Через 80 миллисекунд	Даёт браузеру время отрисовать элементы
updateMaxScroll()	    Через 80мс	            Вычисляет максимальную прокрутку
syncScroll(0)	        Через 80мс	            Устанавливает прокрутку в начало
updateLabelPos()	    Через 80мс	            Устанавливает позицию надписи
bindSync(...)	        Через 80мс	            Связывает блоки для синхронной прокрутки

Почему задержка 80мс:
Браузеру нужно время, чтобы:

    Добавить созданные элементы в DOM

    Вычислить их размеры (scrollWidth, clientWidth)

Без задержки размеры могут быть вычислены неправильно (0 или старые значения).



ПОЛНАЯ СХЕМА РАБОТЫ СКРИПТА

1. Загрузка страницы
   ↓
2. Скрипт запускается (IIFE)
   ↓
3. Поиск контейнеров на странице (.section_Three .reviewsWidget)
   ↓
4. build() — создание карточек и пустых заглушек
   ↓
5. setTimeout(80мс) — ожидание отрисовки
   ↓
6. updateMaxScroll() — расчёт максимальной прокрутки
   ↓
7. syncScroll(0) — установка начальной позиции
   ↓
8. bindSync() — связывание блоков для синхронизации
   ↓
9. Ожидание действий пользователя
   ↓
   ├── Движение ползунка → syncScroll() + updateLabelPos()
   ├── Ручная прокрутка блока → синхронизация второго блока и ползунка
   └── Изменение размера окна → debounce → перерасчёт
